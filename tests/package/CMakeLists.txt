cmake_minimum_required(VERSION 3.25)
project(UniMemoryPackageLookup LANGUAGES CXX)

# Confine dependency discovery to fixtures; the compiler is already configured.
set(CMAKE_FIND_USE_CMAKE_ENVIRONMENT_PATH FALSE)
set(CMAKE_FIND_USE_SYSTEM_ENVIRONMENT_PATH FALSE)
set(CMAKE_FIND_USE_PACKAGE_REGISTRY FALSE)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY FALSE)
set(CMAKE_FIND_USE_CMAKE_SYSTEM_PATH FALSE)
set(CMAKE_PREFIX_PATH "")
set(CMAKE_LIBRARY_PATH "")
set(CMAKE_INCLUDE_PATH "")
set(prefix "${CMAKE_CURRENT_BINARY_DIR}/dependency")
set(UNIMEMORY_JEMALLOC_INCLUDE_DIR UNIMEMORY_JEMALLOC_INCLUDE_DIR-NOTFOUND CACHE PATH "" FORCE)
set(UNIMEMORY_JEMALLOC_RELEASE_LIBRARY UNIMEMORY_JEMALLOC_RELEASE_LIBRARY-NOTFOUND CACHE FILEPATH "" FORCE)
set(UNIMEMORY_JEMALLOC_DEBUG_LIBRARY UNIMEMORY_JEMALLOC_DEBUG_LIBRARY-NOTFOUND CACHE FILEPATH "" FORCE)
if(NOT SCENARIO STREQUAL "missing_include")
    file(WRITE "${prefix}/include/jemalloc/jemalloc.h" "/* discovery fixture */\n")
    set(UNIMEMORY_JEMALLOC_INCLUDE_DIR "${prefix}/include" CACHE PATH "" FORCE)
endif()
set(release "${prefix}/lib/${CMAKE_STATIC_LIBRARY_PREFIX}jemalloc${CMAKE_STATIC_LIBRARY_SUFFIX}")
set(debug "${prefix}/debug/lib/${CMAKE_STATIC_LIBRARY_PREFIX}jemalloc${CMAKE_STATIC_LIBRARY_SUFFIX}")
file(REMOVE "${release}" "${debug}")
if(NOT SCENARIO MATCHES "^(missing_include|missing_library|retry)$")
    file(WRITE "${release}" "valid\n")
    if(NOT SCENARIO STREQUAL "debug_fallback")
        file(WRITE "${debug}" "valid\n")
    endif()
endif()

# Exercise Unix validation on every host, with a controlled executable in place
# of nm. Actual native libraries and their real nm are covered by backend builds.
set(UNIX TRUE)
set(APPLE FALSE)
set(CMAKE_NM "${NM_FIXTURE}")
if(SCENARIO STREQUAL "missing_nm")
    set(CMAKE_NM "")
elseif(SCENARIO STREQUAL "nm_unavailable")
    set(CMAKE_NM "${prefix}/missing-nm")
elseif(SCENARIO STREQUAL "nm_error")
    file(WRITE "${release}" "error\n")
elseif(SCENARIO STREQUAL "release_replaces_new")
    file(WRITE "${release}" "replace\n")
elseif(SCENARIO STREQUAL "debug_replaces_new")
    file(WRITE "${debug}" "replace\n")
endif()

if(LOOKUP_MODE STREQUAL "source")
    set(UNIMEMORY_WITH_MIMALLOC OFF CACHE BOOL "" FORCE)
    set(UNIMEMORY_WITH_JEMALLOC ON CACHE BOOL "" FORCE)
    set(UNIMEMORY_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    add_subdirectory("${UNIMEMORY_SOURCE}" unimemory)
    message(FATAL_ERROR "An explicit unavailable source backend must fail")
endif()

# Generate the shipped package config independently of enabled build backends.
# The target file is a sentinel: failure must return before importing it.
if(INSTALLED_PACKAGE)
    set(package "${INSTALLED_PACKAGE}")
else()
    include(CMakePackageConfigHelpers)
    set(UNIMEMORY_WITH_MIMALLOC OFF)
    set(UNIMEMORY_WITH_JEMALLOC ON)
    set(package "${CMAKE_CURRENT_BINARY_DIR}/sdk/lib/cmake/UniMemory")
    configure_package_config_file("${UNIMEMORY_SOURCE}/cmake/UniMemoryConfig.cmake.in"
        "${package}/UniMemoryConfig.cmake" INSTALL_DESTINATION lib/cmake/UniMemory)
    configure_file("${UNIMEMORY_SOURCE}/cmake/UniMemoryJemalloc.cmake"
        "${package}/UniMemoryJemalloc.cmake" COPYONLY)
    file(WRITE "${package}/UniMemoryTargets.cmake"
        "add_library(UniMemory::UniMemory INTERFACE IMPORTED)\n")
endif()

if(LOOKUP_MODE STREQUAL "required")
    find_package(UniMemory CONFIG REQUIRED PATHS "${package}" NO_DEFAULT_PATH)
else()
    find_package(UniMemory CONFIG QUIET PATHS "${package}" NO_DEFAULT_PATH)
endif()
if(SCENARIO MATCHES "^(valid|debug_fallback)$")
    if(NOT UniMemory_FOUND OR NOT TARGET UniMemory::UniMemory OR NOT TARGET unimemory_jemalloc)
        message(FATAL_ERROR "Valid dependency must import the package")
    endif()
    get_target_property(actual_release unimemory_jemalloc IMPORTED_LOCATION_RELEASE)
    if(NOT actual_release STREQUAL release)
        message(FATAL_ERROR "Wrong release library: ${actual_release}")
    endif()
    if(SCENARIO STREQUAL "debug_fallback")
        get_target_property(actual_debug unimemory_jemalloc MAP_IMPORTED_CONFIG_DEBUG)
        if(NOT actual_debug STREQUAL "Release")
            message(FATAL_ERROR "Missing debug library must map to Release")
        endif()
    else()
        get_target_property(actual_debug unimemory_jemalloc IMPORTED_LOCATION_DEBUG)
        if(NOT actual_debug STREQUAL debug)
            message(FATAL_ERROR "Debug discovery mixed CRT configurations")
        endif()
    endif()
else()
    if(UniMemory_FOUND OR TARGET UniMemory::UniMemory OR TARGET unimemory_jemalloc)
        message(FATAL_ERROR "Unavailable dependency left a found package or partial imported target")
    endif()
    if(NOT UniMemory_NOT_FOUND_MESSAGE)
        message(FATAL_ERROR "Unavailable dependency must provide a diagnostic reason")
    endif()
    if(SCENARIO STREQUAL "retry")
        file(WRITE "${release}" "valid\n")
        # CMake can cache directory listings within one configure. Supply the
        # corrected path explicitly, as a consumer overriding discovery would.
        set(UNIMEMORY_JEMALLOC_RELEASE_LIBRARY "${release}" CACHE FILEPATH "" FORCE)
        find_package(UniMemory CONFIG REQUIRED PATHS "${package}" NO_DEFAULT_PATH)
        if(NOT UniMemory_FOUND OR NOT TARGET UniMemory::UniMemory OR NOT TARGET unimemory_jemalloc)
            message(FATAL_ERROR "A corrected dependency must be discoverable after optional failure")
        endif()
    endif()
endif()
message(STATUS "UniMemory package contract verified")
